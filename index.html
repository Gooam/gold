function selectCandleAtLogicalX(logicalX) { if (!dailyCandles || dailyCandles.length === 0) { selectedCandleIdx = -1; hideTooltip(); renderCandleChart(false); return; } var leftAxisW = 64; var relX = logicalX - leftAxisW + scrollOffsetPx; var idx = Math.floor(relX / candleStepPx); if (idx >= 0 && idx < dailyCandles.length) { if (idx === selectedCandleIdx) { selectedCandleIdx = -1; hideTooltip(); renderCandleChart(false); return; } selectedCandleIdx = idx; showTooltipForCandle(dailyCandles[idx], logicalX); renderCandleChart(false); } else { selectedCandleIdx = -1; hideTooltip(); renderCandleChart(false); } }
